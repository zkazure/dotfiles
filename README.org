* Getting Started

#+begin_src sh
  # 检查 stow，没有则安装（--dotfiles 需要 >= 2.3）
  command -v stow >/dev/null 2>&1 || sudo apt install stow

  # 克隆到 $HOME 下（路径必须能解析到 ~/dotfiles）
  cd ~
  git clone <repo-url> dotfiles

  cd ~/dotfiles
  # .config 内容逐文件链接到 ~/.config
  stow --dotfiles --no-folding -t ~/.config .config
  # 家目录点文件链接到 ~（dot- 前缀映射）
  stow --dotfiles -t ~ home
  # .themes 整目录链接
  ln -s dotfiles/.themes ~/.themes
#+end_src

本地文件（=~/.config/resticprofile/.pw=、=~/.config/xfce4/xfconfbak= =panelbak= =src=）不进仓库，stow 不会碰它们。

* sudo privilege
** mouse
=/etc/X11/xorg.conf.d/50-mouse-acceleration.conf=
#+begin_src conf :tangle no
  Section "InputClass"
      Identifier "My Mouse"
      MatchIsPointer "yes"
      # 强制所有鼠标设备使用 Flat 线性模式，关闭加速
      Option "AccelProfile" "flat"
      # 基础速度乘数，0 为 libinput 默认，有效值域通常为 -1 到 1
      Option "AccelSpeed" "0"
  EndSection
#+end_src

** touchpad
=/etc/X11/xorg.conf.d/01-touchpad.conf=
#+begin_src conf :tangle no
  Section "InputClass"
      Identifier "touchpad"
      MatchIsTouchpad "on"
      Option "Tapping" "on"
      Option "DisableWhileTyping" "on"
      Option "MiddleEmulation" "on"
      Option "NaturalScrolling" "on"
  EndSection
#+end_src
